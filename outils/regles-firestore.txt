rules_version = '2';

// Règles de sécurité de la base Firestore (à coller dans la console Firebase :
// Firestore Database > Règles > remplacer tout le texte > Publier).
//  - Tout le monde peut LIRE les parties.
//  - Un téléphone ne peut créer, modifier ou supprimer QUE ses propres parties
//    (identifiant anonyme donné par Firebase Authentication).
//  - Les données envoyées sont vérifiées (taille, format) pour éviter les abus.
//  - Séries (jours d'affilée) : chaque téléphone écrit la sienne, mais SEUL
//    l'administrateur (connecté avec son compte Google) peut les lire.
//    ⚠ Ce fichier est public : avant de coller, remplacer ADRESSE_ADMIN
//    par l'adresse Gmail de l'administrateur (ne pas l'écrire ici).
service cloud.firestore {
  match /databases/{database}/documents {

    function partieValide() {
      let d = request.resource.data;
      return d.uid == request.auth.uid
        && d.keys().size() <= 15
        && d.joueur is string && d.joueur.size() > 0 && d.joueur.size() <= 40
        && d.parcoursNom is string && d.parcoursNom.size() <= 120
        && d.note is string && d.note.size() <= 2000
        && d.date is string && d.date.size() == 10
        && d.format in ['18', 'aller', 'retour']
        && d.trous is list && d.trous.size() <= 18;
    }

    match /parties/{idPartie} {
      allow read: if true;
      allow create: if request.auth != null
        && idPartie.matches(request.auth.uid + '_[0-9]+')
        && partieValide();
      allow update: if request.auth != null
        && resource.data.uid == request.auth.uid
        && partieValide();
      allow delete: if request.auth != null
        && (resource == null || resource.data.uid == request.auth.uid);
    }

    function serieValide() {
      let d = request.resource.data;
      return d.uid == request.auth.uid
        && d.keys().hasOnly(['uid', 'joueur', 'actuelle', 'record', 'dernierJour', 'majLe'])
        && d.joueur is string && d.joueur.size() > 0 && d.joueur.size() <= 40
        && d.actuelle is int && d.actuelle >= 0 && d.actuelle <= 100000
        && d.record is int && d.record >= 0 && d.record <= 100000
        && d.dernierJour is string && d.dernierJour.size() <= 10;
    }

    // Administrateur : connecté avec un de ses comptes Google
    function estAdmin() {
      return request.auth != null
        && request.auth.token.email in ['ADRESSE_ADMIN_1', 'ADRESSE_ADMIN_2']
        && request.auth.token.email_verified == true;
    }

    match /series/{idAppareil} {
      allow read, delete: if estAdmin();
      allow create, update: if request.auth != null
        && idAppareil == request.auth.uid
        && serieValide();
    }
  }
}
