rules_version = '2';

// Règles de sécurité de la base Firestore (à coller dans la console Firebase :
// Firestore Database > Règles > remplacer tout le texte > Publier).
//  - Parties : tout le monde peut les LIRE ; un téléphone ne peut créer,
//    modifier ou supprimer QUE les siennes.
//  - Séries (jours d'affilée) et scores de putting : chaque joueur écrit les
//    siens, mais SEUL l'administrateur (connecté avec Google) peut les lire.
//  - Sauvegardes : privées, lisibles seulement par leur compte Google.
//  - Les données envoyées sont vérifiées (taille, format) pour éviter les abus.
// ⚠ Ce fichier est public : avant de coller, remplacer ADRESSE_ADMIN_1 et
//   ADRESSE_ADMIN_2 par les adresses Gmail de l'administrateur (ne pas les écrire ici).
service cloud.firestore {
  match /databases/{database}/documents {

    function estAdmin() {
      return request.auth != null
        && request.auth.token.email in ['ADRESSE_ADMIN_1', 'ADRESSE_ADMIN_2']
        && request.auth.token.email_verified == true;
    }

    function estProprietaire(uid) {
      return request.auth != null && uid == request.auth.uid;
    }

    function partieValide() {
      let d = request.resource.data;
      return d.uid == request.auth.uid
        && d.keys().size() <= 15
        && d.joueur is string && d.joueur.size() > 0 && d.joueur.size() <= 40
        && d.parcoursNom is string && d.parcoursNom.size() <= 120
        && d.note is string && d.note.size() <= 2000
        && d.date is string && d.date.size() == 10
        && d.format in ['18', 'aller', 'retour']
        && d.trous is list && d.trous.size() <= 18;
    }

    match /parties/{idPartie} {
      allow read: if true;
      allow create: if request.auth != null
        && idPartie.matches(request.auth.uid + '_[0-9]+')
        && partieValide();
      allow update: if request.auth != null
        && resource.data.uid == request.auth.uid
        && partieValide();
      allow delete: if estAdmin() || (request.auth != null
        && (resource == null || resource.data.uid == request.auth.uid));
    }

    function serieValide() {
      let d = request.resource.data;
      return d.uid == request.auth.uid
        && d.keys().hasOnly(['uid', 'joueur', 'actuelle', 'record', 'dernierJour', 'majLe'])
        && d.joueur is string && d.joueur.size() > 0 && d.joueur.size() <= 40
        && d.actuelle is int && d.actuelle >= 0 && d.actuelle <= 100000
        && d.record is int && d.record >= 0 && d.record <= 100000
        && d.dernierJour is string && d.dernierJour.size() <= 10;
    }

    match /series/{idAppareil} {
      allow read: if estAdmin();
      allow delete: if estAdmin() || estProprietaire(idAppareil);
      allow create, update: if estProprietaire(idAppareil) && serieValide();
    }

    function scoresValides() {
      let d = request.resource.data;
      return d.uid == request.auth.uid
        && d.keys().hasOnly(['uid', 'joueur', 'scores', 'majLe'])
        && d.joueur is string && d.joueur.size() > 0 && d.joueur.size() <= 40
        && d.scores is map && d.scores.size() <= 100;
    }

    match /scores/{idAppareil} {
      allow read: if estAdmin();
      allow delete: if estAdmin() || estProprietaire(idAppareil);
      allow create, update: if estProprietaire(idAppareil) && scoresValides();
    }

    // Sauvegarde privée : seulement avec un compte Google (pas en anonyme)
    function compteGoogle(uid) {
      return estProprietaire(uid) && request.auth.token.firebase.sign_in_provider == 'google.com';
    }

    match /sauvegardes/{idCompte} {
      allow read, delete: if compteGoogle(idCompte);
      allow create, update: if compteGoogle(idCompte)
        && request.resource.data.keys().hasOnly(['uid', 'donnees', 'majLe'])
        && request.resource.data.donnees is string
        && request.resource.data.donnees.size() <= 900000;
    }
  }
}
